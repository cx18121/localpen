# Phase 1.5 Conversational Surface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend the one-shot quick-action panel into a persistent, resumable multi-turn conversation: typing a follow-up under a result promotes it into a paired-bubble thread, threads persist and resume, and refinement chips offer in-panel next steps.

**Architecture:** A dedicated `Conversation` `@Observable` model + `ConversationView`, orchestrated by the existing `PanelController` (the one-shot path is untouched). Each turn is stored as an existing `interactions` row tagged with a new `thread_id`; a new `threads` table holds resume metadata. `StreamingClient` gains a multi-turn `complete(messages:)` entry point.

**Tech Stack:** Swift 5.10, SwiftUI + AppKit (macOS 14), GRDB.swift, XcodeGen (`project.yml` → `.xcodeproj`), XCTest.

**Reference docs:** spec at `docs/superpowers/specs/2026-05-27-phase-1.5-conversational-surface-design.md`; design tokens in `Localpen/DesignSystem/{Tokens,Spacing,Typography}.swift`; existing flow in `Localpen/Panel/PanelController.swift`, `Localpen/Models/{Action,Selection,Interaction,PanelState}.swift`, `Localpen/Services/{Persistence,StreamingClient}.swift`, `Localpen/Models/ActionPrompts.swift`.

**Conventions for every task:** XcodeGen owns the project — after editing `project.yml`, run `xcodegen generate`. The `.xcodeproj` is gitignored, so never `git add` it. Build with:
```
xcodebuild -project Localpen.xcodeproj -scheme Localpen -configuration Debug build
```
Run tests with:
```
xcodebuild test -project Localpen.xcodeproj -scheme Localpen -destination 'platform=macOS'
```
Commit only the files each task names.

```swift
import Foundation

enum ConversationRole: Equatable { case user, assistant }

/// One bubble in the thread. `action` is set on the user turn that was a seed
/// action or a refinement chip; nil for a free-form typed follow-up. Identity
/// is a stable UUID for SwiftUI; equality compares content (so tests are stable
/// across UUIDs).
struct ConversationTurn: Identifiable, Equatable {
    let id = UUID()
    var role: ConversationRole
    var text: String
    var action: Action?

    init(role: ConversationRole, text: String, action: Action? = nil) {
        self.role = role; self.text = text; self.action = action
    }

    static func == (lhs: ConversationTurn, rhs: ConversationTurn) -> Bool {
        lhs.role == rhs.role && lhs.text == rhs.text && lhs.action == rhs.action
    }
}
```

```swift
    /// the associated-value cases). Used to resume a stored thread. Unknown
    /// raw values fall back to `.custom(steer ?? "")`.
    static func reconstruct(rawValue: String, steer: String?) -> Action {
        switch rawValue {
        case "proofread":   return .proofread
        case "polish":      return .polish
        case "more_formal": return .moreFormal
        case "more_casual": return .moreCasual
        case "shorter":     return .shorter
        case "respond":     return .respond(steer ?? "")
        default:            return .custom(steer ?? "")
        }
    }
```

`Localpen/Models/Conversation.swift`:

```swift
import Foundation
import Observation

/// A live, multi-turn refinement thread. Owned by PanelController; ConversationView
/// observes it. `streamingText` holds the in-flight assistant reply so per-token
/// updates re-render only the latest bubble (mirrors PanelController.streamingText).
@MainActor
@Observable
final class Conversation {
    let threadID: Int64
    let originalSelection: Selection
    let operationClass: Action.OperationClass
    var turns: [ConversationTurn]
    var streamingText: String = ""

    init(threadID: Int64, originalSelection: Selection,
         operationClass: Action.OperationClass, turns: [ConversationTurn]) {
        self.threadID = threadID
        self.originalSelection = originalSelection
        self.operationClass = operationClass
        self.turns = turns
    }

    /// Seeds a thread from a just-completed one-shot result plus the first
    /// follow-up: [user(seed action), assistant(result), user(follow-up)].
    static func seeded(threadID: Int64, originalSelection: Selection,
                       seedAction: Action, result: String, followUp: String) -> Conversation {
        Conversation(
            threadID: threadID,
            originalSelection: originalSelection,
            operationClass: seedAction.operationClass,
            turns: [
                ConversationTurn(role: .user, text: seedAction.pillLabel, action: seedAction),
                ConversationTurn(role: .assistant, text: result),
                ConversationTurn(role: .user, text: followUp)
            ]
        )
    }

    /// The OpenAI-style message array (excluding the system message, which the
    /// controller prepends). The first user turn embeds the original selection
    /// via the seed action's user message; later user turns are raw follow-ups.
    var messages: [(role: String, content: String)] {
        turns.enumerated().map { idx, turn in
            switch turn.role {
            case .assistant:
                return ("assistant", turn.text)
            case .user:
                if idx == 0, let action = turn.action {
                    return ("user", ActionPrompts.userMessage(for: action, text: originalSelection.text))
                }
                return ("user", turn.text)
            }
        }
    }
}
```

```swift
import Foundation
import GRDB
import os

/// Resume metadata for the history list.
struct ThreadSummary: Equatable {
    let id: Int64
    let title: String
    let updatedAt: Int64
}

/// Thread persistence over the shared GRDB queue. A thread's turns are stored as
/// `interactions` rows (one per user→assistant exchange) tagged with `thread_id`;
/// `loadTurns` expands each row back into a user turn + an assistant turn.
final class ConversationStore {
    private let dbQueue: DatabaseQueue
    private let log = Logger(subsystem: "app.localpen.Localpen", category: "ConversationStore")

    init(dbQueue: DatabaseQueue) { self.dbQueue = dbQueue }

    /// Inserts a threads row; returns its id. Title is a trimmed selection snippet.
    func createThread(originalSelection: Selection) async throws -> Int64 {
        let now = Int64(Date().timeIntervalSince1970)
        let title = Self.title(from: originalSelection.text)
        return try await dbQueue.write { db in
            try db.execute(sql: """
                INSERT INTO threads (created_at, updated_at, source_app, original_selection, title)
                VALUES (?, ?, ?, ?, ?)
                """, arguments: [now, now, originalSelection.sourceBundleID, originalSelection.text, title])
            return db.lastInsertedRowID
        }
    }

    /// Persists one user→assistant exchange as an interactions row.
    func appendExchange(threadID: Int64, userText: String, action: Action?,
                        modelID: String, assistantText: String) async throws {
        let now = Int64(Date().timeIntervalSince1970)
        var row = Interaction(
            id: nil, timestamp: now,
            sourceApp: nil,
            selection: nil,
            action: action?.rawValue ?? "custom",
            customPrompt: userText,
            modelID: modelID,
            modelOutput: assistantText,
            accepted: false,
            editedOutput: nil,
            threadID: threadID
        )
        try await dbQueue.write { db in
            try row.insert(db)
            try db.execute(sql: "UPDATE threads SET updated_at = ? WHERE id = ?",
                           arguments: [now, threadID])
        }
    }

    func touchThread(threadID: Int64) async throws {
        let now = Int64(Date().timeIntervalSince1970)
        try await dbQueue.write { db in
            try db.execute(sql: "UPDATE threads SET updated_at = ? WHERE id = ?",
                           arguments: [now, threadID])
        }
    }

    func recentThreads(limit: Int) async throws -> [ThreadSummary] {
        try await dbQueue.read { db in
            try Row.fetchAll(db, sql: """
                SELECT id, title, updated_at FROM threads ORDER BY updated_at DESC LIMIT ?
                """, arguments: [limit]).map {
                ThreadSummary(id: $0["id"], title: $0["title"] ?? "", updatedAt: $0["updated_at"])
            }
        }
    }

    /// Expands each stored exchange row into [user turn, assistant turn].
    func loadTurns(threadID: Int64) async throws -> [ConversationTurn] {
        let rows = try await dbQueue.read { db in
            try Row.fetchAll(db, sql: """
                SELECT action, custom_prompt, model_output FROM interactions
                WHERE thread_id = ? ORDER BY timestamp ASC, id ASC
                """, arguments: [threadID])
        }
        var turns: [ConversationTurn] = []
        for (idx, row) in rows.enumerated() {
            let rawAction: String = row["action"] ?? "custom"
            let steer: String? = row["custom_prompt"]
            let model: String = row["model_output"] ?? ""
            if idx == 0 {
                let action = Action.reconstruct(rawValue: rawAction, steer: steer)
                turns.append(ConversationTurn(role: .user, text: action.pillLabel, action: action))
            } else {
                turns.append(ConversationTurn(role: .user, text: steer ?? ""))
            }
            turns.append(ConversationTurn(role: .assistant, text: model))
        }
        return turns
    }

    private static func title(from selection: String) -> String {
        let trimmed = selection.trimmingCharacters(in: .whitespacesAndNewlines)
        let firstLine = trimmed.split(separator: "\n").first.map(String.init) ?? trimmed
        return firstLine.count > 48 ? String(firstLine.prefix(48)) + "…" : firstLine
    }
}
```

```swift
    static func requestBody(messages: [(role: String, content: String)],
                            sampling: ModelDescriptor.Sampling?) -> [String: Any] {
        [
            "model": "local",
            "messages": messages.map { ["role": $0.role, "content": $0.content] },
            "stream": true,
            "temperature": sampling?.temperature ?? 0.4,
            "top_p": sampling?.topP ?? 0.9,
            "repeat_penalty": sampling?.repeatPenalty ?? 1.05
        ]
    }

    func complete(systemPrompt: String, userText: String,
                  sampling: ModelDescriptor.Sampling? = nil) -> AsyncThrowingStream<String, Error> {
        complete(messages: [("system", systemPrompt), ("user", userText)], sampling: sampling)
    }

    func complete(messages: [(role: String, content: String)],
                  sampling: ModelDescriptor.Sampling? = nil) -> AsyncThrowingStream<String, Error> {
        AsyncThrowingStream { continuation in
            let producer = Task {
                do {
                    var request = URLRequest(url: URL(string: "http://127.0.0.1:\(self.port)/v1/chat/completions")!)
                    request.httpMethod = "POST"
                    request.setValue("application/json", forHTTPHeaderField: "Content-Type")
                    request.timeoutInterval = 60
                    request.httpBody = try JSONSerialization.data(
                        withJSONObject: Self.requestBody(messages: messages, sampling: sampling))

                    let (bytes, response) = try await URLSession.shared.bytes(for: request)
                    if let http = response as? HTTPURLResponse, http.statusCode != 200 {
                        throw NSError(domain: "Localpen", code: http.statusCode,
                                      userInfo: [NSLocalizedDescriptionKey: "llama-server returned HTTP \(http.statusCode)"])
                    }
                    for try await line in bytes.lines {
                        if Task.isCancelled { break }
                        guard line.hasPrefix("data: ") else { continue }
                        let payload = String(line.dropFirst(6))
                        if payload == "[DONE]" { break }
                        guard let data = payload.data(using: .utf8),
                              let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
                              let choices = json["choices"] as? [[String: Any]],
                              let delta = choices.first?["delta"] as? [String: Any],
                              let content = delta["content"] as? String else { continue }
                        continuation.yield(content)
                    }
                    continuation.finish()
                } catch is CancellationError {
                    continuation.finish()
                } catch {
                    continuation.finish(throwing: error)
                }
            }
            continuation.onTermination = { _ in producer.cancel() }
        }
    }
```

```swift
    // MARK: - Conversation

    /// Promote the showing one-shot result into a thread on the first follow-up.
    func promoteToConversation(followUp: String) {
        let trimmed = followUp.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty,
              case .visible(let ctx, .result(let result, _)) = state,
              let action = ctx.action, let sel = ctx.selection,
              let session = current else { return }

        // Persisting the seed turn captures the one-shot result as the thread's
        // first exchange; the pending one-shot interaction is now owned by the thread.
        session.pendingInteraction = nil

        Task { @MainActor [weak self] in
            guard let self else { return }
            let threadID = (try? await self.conversationStore?.createThread(originalSelection: sel)) ?? -1
            try? await self.conversationStore?.appendExchange(
                threadID: threadID, userText: action.pillLabel, action: action,
                modelID: self.registry.activeModelID, assistantText: result)
            guard self.current === session else { return }
            let convo = Conversation.seeded(threadID: threadID, originalSelection: sel,
                                            seedAction: action, result: result, followUp: trimmed)
            self.conversation = convo
            self.state = .visible(ctx, .conversing)
            self.runConversationTurn()
        }
    }

    /// Append a typed/chip follow-up to the live thread and stream the next reply.
    func sendFollowUp(_ text: String) {
        let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty, let convo = conversation else { return }
        convo.turns.append(ConversationTurn(role: .user, text: trimmed))
        runConversationTurn()
    }

    /// Tap a refinement chip — a preset follow-up instruction.
    func tapChip(_ instruction: String) { sendFollowUp(instruction) }

    private func runConversationTurn() {
        guard let session = current, let convo = conversation else { return }
        convo.streamingText = ""
        session.cancelStreaming()
        session.streamingTask = Task { @MainActor [weak self] in
            await self?.streamConversation(convo, session: session)
        }
    }

    private func streamConversation(_ convo: Conversation, session: PanelSession) async {
        let messages = [(role: "system", content: ActionPrompts.conversationSystem)] + convo.messages
        let sampling = registry.activeModel?.sampling
        var partial = ""
        // Reuse the inter-token watchdog by reading the existing timeout constant.
        do {
            for try await token in streaming.complete(messages: messages, sampling: sampling) {
                if Task.isCancelled { return }
                partial += token
                convo.streamingText = partial
            }
            if Task.isCancelled { return }
            convo.turns.append(ConversationTurn(role: .assistant, text: partial))
            convo.streamingText = ""
            // The triggering user turn is the last .user turn before this assistant turn.
            let userText = convo.turns.dropLast().last(where: { $0.role == .user })?.text ?? ""
            try? await conversationStore?.appendExchange(
                threadID: convo.threadID, userText: userText, action: nil,
                modelID: registry.activeModelID, assistantText: partial)
        } catch {
            if Task.isCancelled { return }
            // Keep the thread; show the error in the conversation context.
            state = .visible(PanelContext(selection: convo.originalSelection),
                             .error(PanelError(streamingError: error)))
        }
    }

    /// Accept a specific assistant turn: paste back (revise) or copy (create).
    func acceptTurn(_ turn: ConversationTurn) {
        guard turn.role == .assistant, let convo = conversation, let session = current else { return }
        switch convo.operationClass {
        case .create:
            writeToClipboard(turn.text)
            state = .hidden; conversation = nil; current = nil; hidePanelAnimated()
        case .revise:
            let capture = session.capture
            Task { @MainActor [weak self] in
                guard let self else { return }
                let result = await self.textIO.pasteBack(result: turn.text, capture: capture)
                guard self.current === session else { return }
                if result == .skipped {
                    self.writeToClipboard(turn.text)
                }
                self.state = .hidden; self.conversation = nil
                self.current = nil; self.hidePanelAnimated()
            }
        }
    }
```
